general16Sdf = pd.read_csv("/home/daniel/Documents/projects/mossSymbiosis/"
                 "cliffMosses/mossCyano_qPCR/general16S/csv/"
                 "qPCR_2026-04-21_general16S_mosses -  Quantification Summary.csv",
                 index_col="Well", usecols=["Well","Sample","Cq","SQ"]

dilutionDF = general16Sdf.query("Sample == 'Ana'")
x = dilutionDF.SQ.to_numpy().reshape((-1, 1))
y = dilutionDF.Cq.to_numpy()
model = LinearRegression().fit(x, y)
general16S_r2 = model.score(x, y)
print(f"intercept: {model.intercept_}")
print(f"slope: {model.coef_}") ## -10.8658
general16S_slope = round(model.coef_[0], 2)
general16S_interc = round(model.intercept_, 2)
fig, ax = plt.subplots()
ax.plot(x, y, "o")
ax.plot( x, model.predict(x), 
        c='k', linewidth=4, linestyle= "dotted")
ax.set_title("general 16S standard dilution curve")
ax.set_ylabel('cycles till detection (ct)')
ax.set_xlabel('dilution')
plt.text(0.5,20,f"{general16S_slope}x + {general16S_interc}")
plt.text(0.5,18,f"r2={round(general16S_r2,3)}")
